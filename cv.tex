\newcommand{\cvheader}[5]{%
  % #1 Name  #2 Title  #3 Email  #4 Phone  #5 Location
  {\Huge\bfseries #1}\\[0.15em]
  {\large\color{muted} #2}\\[0.5em]
  {\small
    \href{mailto:#3}{#3}
    \enspace\textcolor{muted}{·}\enspace #4
    \enspace\textcolor{muted}{·}\enspace #5
  }
  \vspace{0.8em}
}

\newcommand{\cvlinks}[3]{%
  % #1 LinkedIn URL+label  #2 GitHub URL+label  #3 Portfolio (optional, empty = omit)
  {\small
    \href{https://linkedin.com/in/#1}{linkedin.com/in/#1}%
    \ifx&#3&\else
      \enspace\textcolor{muted}{·}\enspace \href{https://github.com/#2}{github.com/#2}%
      \enspace\textcolor{muted}{·}\enspace \href{https://#3}{#3}%
    \fi
  }
  \vspace{0.4em}
}

  % #1 Role   #2 Company   #3 Dates   #4 Location
  \noindent
  \begin{tabularx}{\linewidth}{@{}X r@{}}

    & \textcolor{muted}{\small #4}
  \end{tabularx}
  \vspace{0.1em}
}
